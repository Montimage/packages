Montimage Packages

Signed APT repository for mmt-dpi, for Ubuntu 22.04, 24.04 and 26.04 and Debian 12 and 13, on amd64 and arm64.

Install

sudo apt-get update
sudo apt-get install -y curl gnupg
sudo install -d -m 0755 /etc/apt/keyrings
keyring=$(mktemp)
curl -fsSL -o "$keyring" https://montimage.github.io/packages/montimage-archive-keyring.gpg
gpg --show-keys "$keyring"   # must show 21630C362624FB205C3D0B184F95AECCFEF48443
sudo install -m 0644 "$keyring" /etc/apt/keyrings/montimage-archive-keyring.gpg
rm -f "$keyring"
. /etc/os-release
printf 'Types: deb\nURIs: https://montimage.github.io/packages\nSuites: %s\nComponents: main\nSigned-By: /etc/apt/keyrings/montimage-archive-keyring.gpg\n' "$VERSION_CODENAME" \
  | sudo tee /etc/apt/sources.list.d/montimage-packages.sources >/dev/null
sudo apt-get update && sudo apt-get install mmt-dpi
